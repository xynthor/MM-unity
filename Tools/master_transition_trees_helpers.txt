    static GameObject MasterTreePrefab(string name,string sourcePath,string barkPath,string leavesPath)
    {
        string path="Assets/World/Enroth/Generated/"+name+".prefab";
        var existing=AssetDatabase.LoadAssetAtPath<GameObject>(path); if(existing) return existing;
        var src=AssetDatabase.LoadAssetAtPath<GameObject>(sourcePath); if(!src) return null;
        var go=(GameObject)PrefabUtility.InstantiatePrefab(src); go.name=name;
        var bark=AssetDatabase.LoadAssetAtPath<Material>(barkPath);
        var leaves=AssetDatabase.LoadAssetAtPath<Material>(leavesPath);
        foreach(var r in go.GetComponentsInChildren<Renderer>(true))
        {
            var old=r.sharedMaterials; var mats=new Material[old.Length];
            for(int i=0;i<old.Length;i++)
            {
                string n=old[i]?old[i].name.ToLowerInvariant():"";
                mats[i]=(n.Contains("leaf")||n.Contains("twig")||n.Contains("needle")||n.Contains("branch"))?leaves:bark;
                if(!mats[i]) mats[i]=old[i];
            }
            r.sharedMaterials=mats; r.shadowCastingMode=ShadowCastingMode.On; r.receiveShadows=true;
        }
        var prefab=PrefabUtility.SaveAsPrefabAsset(go,path); UnityEngine.Object.DestroyImmediate(go); return prefab;
    }

    static void BuildMasterTransitionTrees(Terrain terrain,float gridMinX,float gridMinZ)
    {
        var td=terrain.terrainData;
        var broad=MasterTreePrefab("MasterBroadleaf","Assets/EnvironmentAssets/PolyHaven/TreeSmall02/GameLOD2/tree_small_02_LOD1.fbx","Assets/Materials/NewSorpigal/Environment/TreeSmallBark.mat","Assets/Materials/NewSorpigal/Environment/TreeSmallLeaves.mat");
        var pine=MasterTreePrefab("MasterPine","Assets/Environment/PolyHaven/Models/pine_sapling_small/pine_sapling_small_1k.fbx","Assets/Materials/FrozenHighlands/Environment/PineBark.mat","Assets/Materials/FrozenHighlands/Environment/PineLeaves.mat");
        var banyan=MasterTreePrefab("MasterBanyan","Assets/EnvironmentAssets/UnitySamples/BanyanTree.fbx","Assets/Materials/NewSorpigal/Environment/TreeBark.mat","Assets/Materials/NewSorpigal/Environment/TreeLeaves.mat");
        var protos=new List<TreePrototype>();
        if(broad) protos.Add(new TreePrototype{prefab=broad,bendFactor=.12f});
        if(pine) protos.Add(new TreePrototype{prefab=pine,bendFactor=.08f});
        if(banyan) protos.Add(new TreePrototype{prefab=banyan,bendFactor=.10f});
        if(protos.Count==0) return;
        td.treePrototypes=protos.ToArray();
        var alpha=td.GetAlphamaps(0,0,td.alphamapWidth,td.alphamapHeight);
        var trees=new List<TreeInstance>();
        int aw=alpha.GetLength(1),ah=alpha.GetLength(0);
        for(int iz=0;iz<64;iz++) for(int ix=0;ix<96;ix++)
        {
            float jx=Mathf.PerlinNoise(ix*2.17f+3.4f,iz*.73f+8.2f)-.5f;
            float jz=Mathf.PerlinNoise(ix*.81f+6.7f,iz*2.31f+1.8f)-.5f;
            float nx=Mathf.Clamp01((ix+.5f+jx*.78f)/96f),nz=Mathf.Clamp01((iz+.5f+jz*.78f)/64f);
            float wx=terrain.transform.position.x+nx*td.size.x,wz=terrain.transform.position.z+nz*td.size.z;
            float localH=td.GetInterpolatedHeight(nx,nz),wy=terrain.transform.position.y+localH;
            if(wy<OceanY+.8f) continue;
            int ax=Mathf.Clamp(Mathf.RoundToInt(nx*(aw-1)),0,aw-1),ay=Mathf.Clamp(Mathf.RoundToInt(nz*(ah-1)),0,ah-1);
            float temp=alpha[ay,ax,0],trop=alpha[ay,ax,1],dark=alpha[ay,ax,2],swamp=alpha[ay,ax,3],sand=alpha[ay,ax,4],snow=alpha[ay,ax,5],rock=alpha[ay,ax,6],ash=alpha[ay,ax,7],road=alpha[ay,ax,8];
            if(road>.12f||rock>.72f||sand>.72f||ash>.70f) continue;
            float seam=TransitionStrength((wx-gridMinX)/RegionSize,(wz-gridMinZ)/RegionSize);
            float noise=Mathf.PerlinNoise(wx*.0041f+5.2f,wz*.0037f+9.1f);
            float green=Mathf.Clamp01(temp+trop*.95f+dark*.78f+swamp*.62f);
            float chance=green*(.34f+.66f*seam);
            if(chance<.16f||noise<Mathf.Lerp(.76f,.48f,chance)) continue;
            int proto=0;
            if(snow>.16f||dark>.52f) proto=Mathf.Min(1,protos.Count-1);
            else if(trop>.42f && protos.Count>2) proto=2;
            var ti=new TreeInstance();
            ti.prototypeIndex=proto;
            ti.position=new Vector3(nx,Mathf.Clamp01(localH/td.size.y),nz);
            ti.rotation=noise*Mathf.PI*2f;
            ti.widthScale=Mathf.Lerp(.78f,1.28f,Mathf.PerlinNoise(wx*.011f,wz*.009f));
            ti.heightScale=Mathf.Lerp(.78f,1.35f,Mathf.PerlinNoise(wx*.007f+4f,wz*.012f+2f));
            ti.color=Color.white; ti.lightmapColor=Color.white;
            trees.Add(ti);
        }
        td.SetTreeInstances(trees.ToArray(),true);
        Debug.Log("ENROTH_MASTER_TRANSITION_TREES "+trees.Count);
    }
