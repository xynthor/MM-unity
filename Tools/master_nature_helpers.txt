    static bool NearArchitecture(Vector3 p,float radius)
    {
        foreach(var c in Physics.OverlapSphere(p+Vector3.up*1.2f,radius,~0,QueryTriggerInteraction.Ignore))
        {
            Transform t=c.transform;
            while(t){if(t.name.StartsWith("Architecture -",StringComparison.OrdinalIgnoreCase)) return true;t=t.parent;}
        }
        return false;
    }
    static float AlphaAt(float[,,] a,float nx,float nz,int layer)
    {
        int x=Mathf.Clamp(Mathf.RoundToInt(nx*(a.GetLength(1)-1)),0,a.GetLength(1)-1);
        int y=Mathf.Clamp(Mathf.RoundToInt(nz*(a.GetLength(0)-1)),0,a.GetLength(0)-1);
        return a[y,x,layer];
    }
