    static Material MasterDetailMaterial(string name,Color color)
    {
        string path="Assets/World/Enroth/Generated/"+name+".mat";
        var m=AssetDatabase.LoadAssetAtPath<Material>(path);
        if(!m){m=new Material(Shader.Find("Standard"));AssetDatabase.CreateAsset(m,path);}
        m.color=color; m.SetFloat("_Glossiness",.02f); m.SetFloat("_Metallic",0f);
        m.doubleSidedGI=true; EditorUtility.SetDirty(m); return m;
    }
    static GameObject MasterGrassPrefab(string name,Color color)
    {
        string path="Assets/World/Enroth/Generated/"+name+".prefab";
        var old=AssetDatabase.LoadAssetAtPath<GameObject>(path); if(old) return old;
        var src=AssetDatabase.LoadAssetAtPath<GameObject>("Assets/EnvironmentAssets/UnitySamples/grass_bladeNoLOD10.FBX");
        if(!src) return null;
        var go=(GameObject)PrefabUtility.InstantiatePrefab(src); go.name=name;
        var mat=MasterDetailMaterial(name+"Mat",color);
        foreach(var r in go.GetComponentsInChildren<Renderer>(true)) r.sharedMaterial=mat;
        var prefab=PrefabUtility.SaveAsPrefabAsset(go,path); UnityEngine.Object.DestroyImmediate(go);
        return prefab;
    }
    static void BuildMasterGrassDetails(TerrainData td,float[,,] alpha)
    {
        var lush=MasterGrassPrefab("EnrothLushGrass",new Color(.23f,.48f,.14f));
        var dry=MasterGrassPrefab("EnrothDryGrass",new Color(.48f,.40f,.18f));
        if(!lush||!dry) return;
        td.SetDetailResolution(512,16);
        td.detailPrototypes=new[]{
            new DetailPrototype{prototype=lush,usePrototypeMesh=true,renderMode=DetailRenderMode.VertexLit,minWidth=.55f,maxWidth=1.10f,minHeight=.48f,maxHeight=1.15f,noiseSpread=.32f,healthyColor=new Color(.28f,.58f,.18f),dryColor=new Color(.32f,.39f,.13f)},
            new DetailPrototype{prototype=dry,usePrototypeMesh=true,renderMode=DetailRenderMode.VertexLit,minWidth=.45f,maxWidth=.95f,minHeight=.38f,maxHeight=.90f,noiseSpread=.38f,healthyColor=new Color(.58f,.50f,.24f),dryColor=new Color(.42f,.31f,.13f)}
        };
        int res=512, ar=alpha.GetLength(0); var lushD=new int[res,res]; var dryD=new int[res,res];
        long lushN=0,dryN=0;
        for(int y=0;y<res;y++) for(int x=0;x<res;x++)
        {
            int ax=Mathf.Clamp(Mathf.RoundToInt(x/(float)(res-1)*(ar-1)),0,ar-1);
            int ay=Mathf.Clamp(Mathf.RoundToInt(y/(float)(res-1)*(ar-1)),0,ar-1);
            float temp=alpha[ay,ax,0],trop=alpha[ay,ax,1],dark=alpha[ay,ax,2],swamp=alpha[ay,ax,3];
            float sand=alpha[ay,ax,4],snow=alpha[ay,ax,5],rock=alpha[ay,ax,6],ash=alpha[ay,ax,7],road=alpha[ay,ax,8];
            float n=Mathf.PerlinNoise(x*.087f+4.1f,y*.079f+7.6f);
            if(road>.12f||rock>.62f||snow>.50f) continue;
            float green=Mathf.Clamp01(temp+trop*.95f+dark*.55f+swamp*.72f-sand*.72f-ash*.45f);
            float drySuit=Mathf.Clamp01(sand*.70f+ash*.52f+dark*.18f-green*.25f);
            int ld=green>.16f?Mathf.Clamp(Mathf.RoundToInt((green*(.55f+.45f*n))*6f),0,6):0;
            int dd=drySuit>.10f?Mathf.Clamp(Mathf.RoundToInt((drySuit*(.45f+.55f*(1f-n)))*4f),0,4):0;
            lushD[y,x]=ld; dryD[y,x]=dd; lushN+=ld; dryN+=dd;
        }
        td.SetDetailLayer(0,0,0,lushD); td.SetDetailLayer(0,0,1,dryD);
        Debug.Log($"ENROTH_MASTER_GRASS lush={lushN} dry={dryN} resolution={res}");
    }
